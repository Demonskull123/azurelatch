local ShimSection = {}
ShimSection.__index = ShimSection
function ShimSection.new(parent) return setmetatable({ _p = parent }, ShimSection) end
function ShimSection:Section(name)
    if self._p.Section then pcall(self._p.Section, self._p, { Title = name, Box = true, BoxBorder = true, Opened = true }) end
    return self
end
function ShimSection:Label(text)
    if self._p.Section then pcall(self._p.Section, self._p, { Title = tostring(text), TextSize = 14 }) end
    return self
end
function ShimSection:Toggle(name, default, cb)
    local created
    if self._p.Toggle then
        created = self._p:Toggle({ Title = name, Value = default and true or false, Callback = cb or noop })
    end
    return created or self
end
function ShimSection:Slider(name, min, max, default, cb, suffix)
    if self._p.Slider then
        self._p:Slider({ Title = name, Step = 1, Value = { Min = min, Max = max, Default = default }, Callback = cb or noop })
    end
    return self
end
function ShimSection:Dropdown(name, list, default, cb)
    if self._p.Dropdown then
        self._p:Dropdown({ Title = name, Values = list, Value = default, Callback = cb or noop })
    end
    return self
end
function ShimSection:Button(name, arg2, arg3)
    local cb = arg3 or arg2
    if type(arg2) == "function" then cb = arg2 end
    if self._p.Button then
        self._p:Button({ Title = name, Justify = "Center", Callback = cb or noop })
    end
    return self
end
function ShimSection:Textbox(name, placeholder, cb)
    if self._p.Input then
        self._p:Input({ Title = name, Placeholder = placeholder, Value = "", Callback = cb or noop })
    end
    return self
end
function ShimSection:Stat(name, default)
    local value = default or "0"
    local obj = {}
    if self._p.Section then
        local lbl = self._p:Section({ Title = name .. ": " .. value, TextSize = 14 })
        function obj.Set(v)
            value = tostring(v)
            if lbl and lbl.SetTitle then pcall(lbl.SetTitle, lbl, name .. ": " .. value) end
        end
    else
        function obj.Set(v) value = tostring(v) end
    end
    return obj
end

local function playersOnPitch(wantEnemy)
    local me, out = myChar(), {}
    local mine = teamOf(me)
    local chars = Workspace:FindFirstChild("characters")
    for _, ch in ipairs(chars and chars:GetChildren() or {}) do
        local t = teamOf(ch)
        local r = ch:FindFirstChild("HumanoidRootPart")
        if ch ~= me and r and (t == "A" or t == "B") and ((t ~= mine) == wantEnemy) then out[#out + 1] = ch end
    end
    return out
end
local function nearest(list, from)
    local best, bd
    for _, ch in ipairs(list) do
        local d = (ch.HumanoidRootPart.Position - from).Magnitude
        if not bd or d < bd then best, bd = ch, d end
    end
    return best, bd
end

local Aim = { pass = nil, shots = 0, last = "-" }
local function aimPoint()
    local team = teamOf(myChar())
    local goal = Workspace.map:FindFirstChild(team == "A" and "Bgoal" or "Agoal")
    if not goal then return end
    local gk = Workspace:FindFirstChild("gks") and Workspace.gks:FindFirstChild(team == "A" and "B" or "A")
    local gr = gk and gk:FindFirstChild("HumanoidRootPart")
    local half = goal.Size.X / 2 - 9
    local side = 0
    if Config.AimSpot == "Left Corner" then side = -1
    elseif Config.AimSpot == "Right Corner" then side = 1
    elseif Config.AimSpot == "Away From Keeper" then side = (gr and gr.Position.X > goal.Position.X) and -1 or 1 end
    return goal.Position + Vector3.new(side * half, 0, 0)
end
local function shotPowerValue()
    local p = math.clamp(tonumber(Config.ShotPower) or 5, 1, 10)
    return math.floor(29 + ((p - 1) / 9) * (95 - 29))
end
local function rewriteKick(a)
    local r = myRoot()
    if not r then return end
    if Aim.pass and os.clock() - Aim.pass.at < 1 and Aim.pass.char.Parent then
        local to = Aim.pass.char.HumanoidRootPart.Position - r.Position
        a[2] = Config.PassPower
        a[3] = false
        a[4] = Vector3.new(to.X, to.Magnitude * 0.04, to.Z).Unit
        Aim.pass = nil
        Aim.last = "pass"
        return
    end
    if not Config.Aimbot then return end
    local tp = aimPoint()
    if not tp then return end
    local flat = Vector3.new(tp.X - r.Position.X, 0, tp.Z - r.Position.Z).Unit
    a[4] = Vector3.new(flat.X, (tonumber(Config.AimHeight) or 12) / 100, flat.Z).Unit
    if Config.FullPower then
        a[2] = 95
    else
        a[2] = shotPowerValue()
    end
    if Config.NoCurve then a[3] = false end
    Aim.shots += 1
    Aim.last = Config.AimSpot
end

        if HubAlive and type(a) == "table" and a[1] == "kick" then pcall(rewriteKick, a) end

local function kickNow()
    local b = client()
    if not b or b.kickactived or b.moveactived then return false end
    task.spawn(pcall, b.use, b, "kick")
    task.delay(0.12, function()
        b.moveactived = true
        task.wait()
        b.moveactived = false
    end)
    return true
end

local REACH0 = wiki and wiki.ball_pickups and { player = wiki.ball_pickups.player, gk = wiki.ball_pickups.gk }
local function applyReach()
    if not (wiki and wiki.ball_pickups and REACH0) then return end
    local cap = tonumber(wiki.ball_pickups.max) or 25
    if Config.Reach and HubAlive then
        local v = math.clamp(tonumber(Config.ReachSize) or 25, REACH0.player, cap)
        wiki.ball_pickups.player = v
        wiki.ball_pickups.gk = math.max(v, REACH0.gk)
    else
        wiki.ball_pickups.player, wiki.ball_pickups.gk = REACH0.player, REACH0.gk
    end
end

local Auto = { tackleAt = 0, dribAt = 0, passAt = 0, shootAt = 0, gap = setmetatable({}, { __mode = "k" }) }
track(RunService.Heartbeat:Connect(function()
    if not HubAlive then return end
    applyReach()
    local r = myRoot()
    if not (r and onPitch()) then return end
    local now = os.clock()
    local mine = iHaveBall()

    if mine and Config.AutoShoot and now - Auto.shootAt > 1.5 then
        local tp = aimPoint()
        if tp and (tp - r.Position).Magnitude <= Config.ShootRange then
            Auto.shootAt = now
            if kickNow() then return end
        end
    end

    if mine then
        local foe, fd = nearest(playersOnPitch(true), r.Position)
        local closing = 0
        if foe then
            local prev = Auto.gap[foe]
            if prev and now > prev.t then closing = (prev.d - fd) / (now - prev.t) end
            Auto.gap[foe] = { d = fd, t = now }
        end
        local coming = foe and (foe:GetAttribute("Tackling") == true or closing > 14 or fd <= 4)
        if foe and coming then
            if Config.AutoPass and fd <= Config.DribbleRange + 4 and now - Auto.passAt > 1.5 then
                local mate = nearest(playersOnPitch(false), r.Position)
                if mate then
                    Auto.passAt = now
                    Aim.pass = { char = mate, at = now }
                    kickNow()
                    return
                end
            end
            if Config.AutoDribble and fd <= Config.DribbleRange and now - Auto.dribAt > 0.6 then
                Auto.dribAt = now
                useMove("drib")
            end
        end
    elseif Config.AutoTackle and now - Auto.tackleAt > 0.8 then
        local h = holder()
        if h and h.Parent and teamOf(h) ~= teamOf(myChar()) and (teamOf(h) == "A" or teamOf(h) == "B") then
            local hr = h:FindFirstChild("HumanoidRootPart")
            if hr and (hr.Position - r.Position).Magnitude <= Config.TackleRange then
                Auto.tackleAt = now
                local flat = Vector3.new(hr.Position.X, r.Position.Y, hr.Position.Z)
                r.CFrame = CFrame.lookAt(r.Position, flat)
                useMove("dash")
            end
        end
    end
end))

local function disableCollisions()
    local map = Workspace:FindFirstChild("map")
    if not map then return end
    local gk = map:FindFirstChild("gkbarriar")
    if gk then
        if gk:FindFirstChild("A") then gk.A.CanCollide = false end
        if gk:FindFirstChild("B") then gk.B.CanCollide = false end
    end
    local ag, bg = map:FindFirstChild("Agoal"), map:FindFirstChild("Bgoal")
    if ag then ag.CanCollide = false end
    if bg then bg.CanCollide = false end
end
local function setCollisions(on)
    local map = Workspace:FindFirstChild("map")
    if not map then return end
    local gk = map:FindFirstChild("gkbarriar")
    if gk then
        if gk:FindFirstChild("A") then gk.A.CanCollide = on end
        if gk:FindFirstChild("B") then gk.B.CanCollide = on end
    end
    local ag, bg = map:FindFirstChild("Agoal"), map:FindFirstChild("Bgoal")
    if ag then ag.CanCollide = on end
    if bg then bg.CanCollide = on end
end

local autoGoalState = { enabled = false, conn = nil }
local function startAutoGoal()
    if autoGoalState.enabled then return end
    autoGoalState.enabled = true
    local map = Workspace:WaitForChild("map")
    local agoal, bgoal = map:WaitForChild("Agoal"), map:WaitForChild("Bgoal")
    autoGoalState.conn = RunService.RenderStepped:Connect(function()
        if not autoGoalState.enabled then return end
        pcall(function()
            if not onPitch() then return end
            disableCollisions()
            local r = myRoot()
            if not r then return end
            local ball = Workspace.Terrain:FindFirstChild("Ball")
            if ball and not holder() then
                r.CFrame = CFrame.new(ball.Position.X, 0, ball.Position.Z)
            end
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local ob = p.Character:FindFirstChild("Ball")
                    local oroot = p.Character:FindFirstChild("HumanoidRootPart")
                    if ob and oroot and r then
                        r.CFrame = ob.CFrame
                        fireBase({{"tackle"}})
                    end
                end
            end
            if iHaveBall() then
                local team = teamOf(myChar())
                local goal = team == "A" and bgoal or agoal
                if goal and r then
                    r.CFrame = goal.CFrame
                    task.wait(0.185)
                    fireBase({{"kick", 20, false, vector.create(0,1,0)}})
                end
            end
        end)
    end)
end
local function stopAutoGoal()
    autoGoalState.enabled = false
    if autoGoalState.conn then autoGoalState.conn:Disconnect() autoGoalState.conn = nil end
end

    autoGkState.look.Parent = hrp

    local ballHolder = RSW and RSW:FindFirstChild("ballHolder")
    if not ballHolder then return end
    autoGkState.lastHolder = ballHolder.Value

    local velSmooth, lastVel = Vector3.zero, Vector3.zero
    local GRAVITY = Vector3.new(0, -Workspace.Gravity, 0)
    local magnetBurstActive = false

    local function getBall()
        local b = Workspace.Terrain:FindFirstChild("Ball")
        if not b then b = Workspace:FindFirstChild("Ball", true) end
        return b
    end

    local function predictgoalimpact(pos, vel, accel, goal)
        if not goal then return nil end
        local cf = goal.CFrame
        local normal = cf.LookVector
        local rel = pos - cf.Position
        local a = 0.5 * accel:Dot(normal)
        local b = vel:Dot(normal)
        local c = rel:Dot(normal)
        if math.abs(a) < 1e-6 then
            if math.abs(b) < 1e-6 then return nil end
            local t = -c / b
            return t > 0 and t or nil
        end
        local disc = b*b - 4*a*c
        if disc < 0 then return nil end
        local sd = math.sqrt(disc)
        local t1 = (-b - sd) / (2*a)
        local t2 = (-b + sd) / (2*a)
        return (t1 > 0 and t1) or (t2 > 0 and t2)
    end

    table.insert(autoGkState.connections, ballHolder.Changed:Connect(function(cur)
        if autoGkState.lastHolder == char and cur == nil then
            autoGkState.pause = true
            task.delay(2.6, function() autoGkState.pause = false end)
        end
        autoGkState.lastHolder = cur
    end))

    table.insert(autoGkState.connections, RunService.RenderStepped:Connect(function(dt)
        if not autoGkState.enabled or autoGkState.pause then
            if autoGkState.look then autoGkState.look.Enabled = false end
            return
        end
        if not onPitch() then
            if autoGkState.look then autoGkState.look.Enabled = false end
            return
        end
        if not (hrp and hrp.Parent) then return end
        local currentHolder = ballHolder.Value
        if currentHolder == char then
            if autoGkState.look then autoGkState.look.Enabled = false end
            return
        end
        local ball = getBall()
        if currentHolder and currentHolder:FindFirstChild("HumanoidRootPart") then
            local hr = currentHolder.HumanoidRootPart
            if autoGkState.look then
                autoGkState.look.Enabled = true
                autoGkState.look.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(hr.Position.X, hrp.Position.Y, hr.Position.Z))
            end
            if (hrp.Position - hr.Position).Magnitude <= 35 then
                fireBase({{"tackle"}})
            end
        elseif ball then
            local rawVel = ball.AssemblyLinearVelocity
            velSmooth = velSmooth:Lerp(rawVel, math.clamp(dt*15, 0, 1))
            local accel = (velSmooth - lastVel) / math.max(dt, 1/240)
            lastVel = velSmooth
            local team = teamOf(char)
            local map = Workspace:FindFirstChild("map")
            local goal = map and map:FindFirstChild(team == "A" and "Agoal" or "Bgoal")
            local tImpact = predictgoalimpact(ball.Position, velSmooth, accel, goal)
            local predicted
            if tImpact then
                predicted = ball.Position + velSmooth * tImpact + 0.5 * accel * tImpact * tImpact
            else
                predicted = ball.Position + velSmooth * 0.25 + 0.5 * GRAVITY * 0.0625
            end
            if autoGkState.look then
                autoGkState.look.Enabled = true
                autoGkState.look.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(predicted.X, hrp.Position.Y, predicted.Z))
            end
            local dist = (hrp.Position - ball.Position).Magnitude
            if dist > 35 and dist <= 37 and ball.Position.Y > hrp.Position.Y + 4 then
                hum.Jump = true
            end
            if dist <= 35 then
                fireBase({{"tackle"}})
                if not magnetBurstActive then
                    magnetBurstActive = true
                    task.spawn(function()
                        local start = os.clock()
                        while os.clock() - start < 0.897 and autoGkState.enabled do
                            fireRaw(buffers.grabball)
                            task.wait(0.125)
                        end
                        magnetBurstActive = false
                    end)
                end
            end
        else
            if autoGkState.look then autoGkState.look.Enabled = false end
        end
    end))

    table.insert(autoGkState.connections, hum.Died:Connect(stopAutoGK))
    table.insert(autoGkState.connections, LocalPlayer.CharacterAdded:Connect(stopAutoGK))
end

local kickHard = false
local kickHookInstalled = false
local function installKickHook()
    if kickHookInstalled then return end
    kickHookInstalled = true
    local ok = pcall(function()
        local meta = getrawmetatable(game)
        local oldIndex = meta.__index
        local oldNamecall = meta.__namecall
        setreadonly(meta, false)
        local function modKick(args)
            if not kickHard then return end
            if type(args) == "table" and type(args[2]) == "table" then
                for _, sub in pairs(args[2]) do
                    if type(sub) == "table" and sub[1] == "kick" then sub[2] = 100 end
                end
            end
        end
        meta.__index = newcclosure(function(self, key)
            if self == ByteNetReliable and (key == "FireServer" or key == "fireServer") then
                return newcclosure(function(remote, ...)
                    local args = {...}
                    modKick(args)
                    return oldIndex(remote, key)(remote, unpack(args))
                end)
            end
            return oldIndex(self, key)
        end)
        meta.__namecall = newcclosure(function(self, ...)
            local args = {...}
            local method = getnamecallmethod()
            if self == ByteNetReliable and (method == "FireServer" or method == "fireServer") then
                modKick(args)
                return oldNamecall(self, unpack(args))
            end
            return oldNamecall(self, ...)
        end)
        setreadonly(meta, true)
    end)
    if not ok then kickHookInstalled = false end
end

local ActiveDashes = {}
local renderConn
local ZERO = Vector3.new()
local DEFAULT_DIR = Vector3.new(0,0,1)
local DirectionOffsets = {
    Front = Vector3.new(0, 0, 1), Back = Vector3.new(0, 0, -1),
    Left = Vector3.new(-1, 0, 0), Right = Vector3.new(1, 0, 0),
    FrontLeft = Vector3.new(-1, 0, 1).Unit, FrontRight = Vector3.new(1, 0, 1).Unit,
    LeftBack = Vector3.new(-1, 0, -1).Unit, RightBack = Vector3.new(1, 0, -1).Unit,
}
local function updateDashes(dt)
    for i = #ActiveDashes, 1, -1 do
        local dash = ActiveDashes[i]
        local hrp = dash.hrp
        if not hrp or not hrp.Parent then
            table.remove(ActiveDashes, i)
        else
            local step = dash.cfg.isUnlimited and (dash.speed * dt) or math.min(dash.speed * dt, dash.remaining)
            local moveDir
            if dash.isDirectional then
                local baseLook = hrp.CFrame.LookVector
                local right = baseLook:Cross(Vector3.new(0,1,0))
                local offset = dash.offset
                moveDir = (right * -offset.X + baseLook * offset.Z).Unit
            else
                local look = hrp.CFrame.LookVector
                moveDir = Vector3.new(look.X, 0, look.Z).Unit
                if moveDir.Magnitude < 0.01 then moveDir = DEFAULT_DIR end
            end
            local newPos = hrp.Position + moveDir * step
            hrp.CFrame = CFrame.new(newPos.X, hrp.Position.Y, newPos.Z) * CFrame.new(ZERO, moveDir)
            dash.remaining = dash.remaining - step
            if dash.remaining <= 0 then table.remove(ActiveDashes, i) end
        end
    end
    if #ActiveDashes == 0 and renderConn then renderConn:Disconnect() renderConn = nil end
end
local function startUpdateIfNeeded()
    if not renderConn and #ActiveDashes > 0 then
        renderConn = RunService.RenderStepped:Connect(updateDashes)
    end
end
local function startDash(hrp, cfg, trackAnim, directionName)
    local dist = Distances[cfg.sliderVar]
    if dist <= 0 then return end
    task.delay(cfg.wait or 0, function()
        if not hrp or not hrp.Parent then return end
        local dash = { hrp = hrp, remaining = dist, cfg = cfg, isDirectional = cfg.isDirectional == true }
        if dash.isDirectional and directionName then
            dash.offset = DirectionOffsets[directionName] or Vector3.new(0,0,1)
        end
        if cfg.isUnlimited then
            dash.remaining = math.huge
            dash.speed = cfg.speed
            if trackAnim then
                local stop
                stop = trackAnim.Stopped:Connect(function() dash.remaining = 0 if stop then stop:Disconnect() end end)
            end
        else
            dash.speed = dist / cfg.duration
        end
        table.insert(ActiveDashes, dash)
        startUpdateIfNeeded()
    end)
end

local lineUpPads, lineUpTargets = {}, {}
local holdingRMB = false
local camLockConn
local activeLineupTarget
local lineupLockActive = false
local currentLineupMode = "Freeze"

local function clearAllPads()
    for _, p in ipairs(lineUpPads) do if p and p.Parent then p:Destroy() end end
    for _, t in ipairs(lineUpTargets) do if t and t.Parent then t:Destroy() end end
    lineUpPads, lineUpTargets = {}, {}
end
local function makePad(pos, col)
    local p = Instance.new("Part")
    p.Name = "LineupPad"
    p.Size = Vector3.new(4,4,4)
    p.Position = pos
    p.Transparency = 0.65
    p.Anchored = true
    p.CanCollide = false
    p.Color = col
    p.Material = Enum.Material.SmoothPlastic
    p.Parent = Workspace
    table.insert(lineUpPads, p)
end
local function makeTarget(pos, col)
    local t = Instance.new("Part")
    t.Name = "LineupTarget"
    t.Shape = Enum.PartType.Ball
    t.Size = Vector3.new(7,7,7)
    t.Position = pos
    t.Anchored = true
    t.CanCollide = false
    t.Color = col
    t.Material = Enum.Material.Neon
    t.Parent = Workspace
    table.insert(lineUpTargets, t)
end
local function getPadUnderPlayer()
    local r = myRoot()
    if not r or not myChar() then return nil end
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {myChar()}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local res = Workspace:Raycast(r.Position, Vector3.new(0, -6, 0), params)
    if res and res.Instance then return res.Instance end
end
local function startCamLock(target)
    activeLineupTarget = target
    if camLockConn then camLockConn:Disconnect() end
    camLockConn = RunService.RenderStepped:Connect(function()
        if not activeLineupTarget then
            if camLockConn then camLockConn:Disconnect() camLockConn = nil end
            return
        end
        Workspace.CurrentCamera.CFrame = CFrame.new(Workspace.CurrentCamera.CFrame.Position, activeLineupTarget.Position)
    end)
end
local function resetLineupPhysics()
    if not lineupLockActive then return end
    lineupLockActive = false
    local char = myChar()
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local r = myRoot()
    if currentLineupMode == "Freeze" and r then r.Anchored = false end
    if currentLineupMode == "Slowness" and hum then hum.WalkSpeed = 30 end
end
local function applyLineupMode()
    lineupLockActive = true
    local char = myChar()
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local r = myRoot()
    if currentLineupMode == "Freeze" and r then r.Anchored = true end
    if currentLineupMode == "Slowness" and hum then hum.WalkSpeed = 5 end
end
track(UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.UserInputType == Enum.UserInputType.MouseButton2 then holdingRMB = true end
end))
track(UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        holdingRMB = false
        activeLineupTarget = nil
        if camLockConn then camLockConn:Disconnect() camLockConn = nil end
        resetLineupPhysics()
    end
end))
track(RunService.RenderStepped:Connect(function()
    if not holdingRMB then resetLineupPhysics() return end
    local pad = getPadUnderPlayer()
    if not pad or pad.Name ~= "LineupPad" then
        resetLineupPhysics()
        activeLineupTarget = nil
        if camLockConn then camLockConn:Disconnect() camLockConn = nil end
        return
    end
    local idx = table.find(lineUpPads, pad)
    if not idx then return end
    local t = lineUpTargets[idx]
    if not t then resetLineupPhysics() return end
    if activeLineupTarget ~= t then startCamLock(t) end
    applyLineupMode()
end))

local function loadLineUps(entries)
    local map = Workspace:WaitForChild("map", 8)
    if not map then return end
    local agoal, bgoal = map:FindFirstChild("Agoal"), map:FindFirstChild("Bgoal")
    if not (agoal and bgoal) then return end
    local colors = {
        Color3.fromRGB(255,0,0), Color3.fromRGB(0,255,0), Color3.fromRGB(255,255,0),
        Color3.fromRGB(0,0,255), Color3.fromRGB(255,0,255), Color3.fromRGB(255,182,193),
    }
    local counts = { A = 0, B = 0 }
    for _, entry in ipairs(entries) do
        local base = entry.base == "A" and agoal.Position or bgoal.Position
        local col = colors[(counts[entry.base] % #colors) + 1]
        counts[entry.base] = counts[entry.base] + 1
        makePad(base + entry.pad, col)
        makeTarget(base + entry.target, col)
    end
end

local function copyLineupCode()
    local char = myChar()
    if not (char and char:FindFirstChild("HumanoidRootPart")) then return end
    local r = char.HumanoidRootPart
    local cam = Workspace.CurrentCamera
    local vp = cam.ViewportSize
    local ray = cam:ViewportPointToRay(vp.X / 2, vp.Y / 2)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {char}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local res = Workspace:Raycast(ray.Origin, ray.Direction * 5000, params)
    if not res then return end
    local map = Workspace:FindFirstChild("map")
    if not map then return end
    local ag, bg = map:FindFirstChild("Agoal"), map:FindFirstChild("Bgoal")
    if not (ag and bg) then return end
    local dA = (r.Position - ag.Position).Magnitude
    local dB = (r.Position - bg.Position).Magnitude
    local useA = dA <= dB
    local goalPos = useA and ag.Position or bg.Position
    local to = res.Position - goalPos
    local po = r.Position - goalPos
    local function rnd(n) return math.round(n) end
    local code = string.format("t(%d,%d,%d)p(%d,%d,%d)%s",
        rnd(to.X), rnd(to.Y), rnd(to.Z), rnd(po.X), rnd(po.Y), rnd(po.Z), useA and "a" or "b")
    setclipboard(code)
    StarterGui:SetCore("SendNotification", { Title = "Lineup Code", Text = code, Duration = 5 })
end
local function loadLineupCodes(text)
    clearAllPads()
    local map = Workspace:FindFirstChild("map")
    if not map then return end
    local ag, bg = map:FindFirstChild("Agoal"), map:FindFirstChild("Bgoal")
    if not (ag and bg) then return end
    local colors = {
        Color3.fromRGB(255,0,0), Color3.fromRGB(255,128,0), Color3.fromRGB(255,255,0),
        Color3.fromRGB(0,255,0), Color3.fromRGB(0,0,255), Color3.fromRGB(128,0,255),
        Color3.fromRGB(255,0,255), Color3.fromRGB(128,255,0), Color3.fromRGB(120,72,0),
        Color3.fromRGB(255,255,255), Color3.fromRGB(0,0,0),
    }
    local i = 0
    for tx, ty, tz, px, py, pz, gl in text:gmatch("t%((%-?[%d%.]+),(%-?[%d%.]+),(%-?[%d%.]+)%)p%((%-?[%d%.]+),(%-?[%d%.]+),(%-?[%d%.]+)%)([ab])") do
        local goalPos = gl == "a" and ag.Position or bg.Position
        local off_t = Vector3.new(tonumber(tx), tonumber(ty), tonumber(tz))
        local off_p = Vector3.new(tonumber(px), tonumber(py) - 3, tonumber(pz))
        local col = colors[(i % #colors) + 1]
        makePad(goalPos + off_p, col)
        makeTarget(goalPos + off_t, col)
        i = i + 1
    end
    StarterGui:SetCore("SendNotification", { Title = "Lines Loaded", Text = tostring(i) .. " created.", Duration = 5 })
end

local goalFarm = { enabled = false, mode = "Solo Farm", stopAt = nil, conn = nil, cfConn = nil, lastTeam = "B", switching = false, lastReset = 0 }
local function checkScoreCondition()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    local hb = pg and pg:FindFirstChild("Hotbar")
    local h = hb and hb:FindFirstChild("Home")
    local a = hb and hb:FindFirstChild("Away")
    if h and a then
        local hg = tonumber(string.match(h.Text, ":%s*(%d+)")) or 0
        local ag = tonumber(string.match(a.Text, "(%d+)%s*:")) or 0
        if math.abs(hg - ag) >= 4 and not (hg >= 10 and ag >= 10) then
            if os.clock() - goalFarm.lastReset >= 60 then return true end
        end
    end
    return false
end
local function isInGame()
    local c = myChar()
    local s = c and c:FindFirstChild("state")
    local ig = s and s:FindFirstChild("ingame")
    return ig and ig.Value or false
end
local function goalFarmStep()
    if not goalFarm.enabled then return end
    if not pick then return end
    local r = myRoot()
    if r then
        if (r.Position - Vector3.new(-371,13,-1599)).Magnitude <= 15 or
           (r.Position - Vector3.new(-196,13,-1599)).Magnitude <= 15 then
            if goalFarm.mode == "Solo Farm" then
                if goalFarm.switching then return end
                if checkScoreCondition() then
                    goalFarm.switching = true
                    goalFarm.lastReset = os.clock()
                    task.spawn(function()
                        local ch = myChar()
                        if ch then
                            local hum = ch:FindFirstChildOfClass("Humanoid")
                            if hum then hum.Health = 0 end
                        end
                        task.wait(8)
                        goalFarm.lastTeam = (goalFarm.lastTeam == "A") and "B" or "A"
                        fireRaw(pick .. "\001\001\000" .. goalFarm.lastTeam)
                        goalFarm.switching = false
                    end)
                    return
                end
                fireRaw(pick .. "\001\001\000" .. goalFarm.lastTeam)
            else
                fireRaw(pick .. "\001\001\000A")
                fireRaw(pick .. "\001\001\000B")
            end
        end
    end
    if not isInGame() then return end
    disableCollisions()
    local ball = Workspace.Terrain:FindFirstChild("Ball")
    if ball and not holder() and r then
        task.wait(0.5)
        r.CFrame = CFrame.new(ball.Position.X, 0, ball.Position.Z)
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local ob = p.Character:FindFirstChild("Ball")
            local oroot = p.Character:FindFirstChild("HumanoidRootPart")
            if ob and oroot and r then
                task.wait(0.5)
                r.CFrame = ob.CFrame
                fireBase({{"tackle"}})
            end
        end
    end
    if iHaveBall() then
        local team = teamOf(myChar())
        local map = Workspace:FindFirstChild("map")
        local goal = map and map:FindFirstChild(team == "A" and "Bgoal" or "Agoal")
        if goal and r then
            task.wait(0.75)
            r.CFrame = goal.CFrame
            task.wait(0.175)
            fireBase({{"kick", 50, false, vector.create(0,1,0)}})
        end
    end
end
local function startGoalFarm()
    if goalFarm.enabled then return end
    goalFarm.enabled = true
    goalFarm.cfConn = LocalPlayer.CharacterAdded:Connect(function()
        task.wait(1)
        if goalFarm.enabled then pcall(goalFarmStep) end
    end)
    track(goalFarm.cfConn)
    goalFarm.conn = RunService.RenderStepped:Connect(function() pcall(goalFarmStep) end)
    task.spawn(function()
        local ls = LocalPlayer:WaitForChild("leaderstats", 10)
        local goals = ls and ls:FindFirstChild("goals")
        while goalFarm.enabled and goals do
            task.wait(1)
            if goalFarm.stopAt and goals.Value >= goalFarm.stopAt then
                goalFarm.enabled = false
                if goalFarm.conn then goalFarm.conn:Disconnect() goalFarm.conn = nil end
                StarterGui:SetCore("SendNotification", { Title = "Goal Farm", Text = "Target reached.", Duration = 3 })
                return
            end
        end
    end)
end
local function stopGoalFarm()
    goalFarm.enabled = false
    if goalFarm.conn then goalFarm.conn:Disconnect() goalFarm.conn = nil end
    if goalFarm.cfConn then goalFarm.cfConn:Disconnect() goalFarm.cfConn = nil end
end

track(RunService.RenderStepped:Connect(function()

        LocalPlayer:Kick("Disconnected.")
    end
end))

local function playEmote(animId, soundId, soundProps, loop)
    local char = myChar()
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local animator = hum and hum:FindFirstChildOfClass("Animator")
    if not (hum and animator) then return end
    local anim = Instance.new("Animation")
    anim.AnimationId = animId
    local trackAnim = animator:LoadAnimation(anim)
    trackAnim:Play()
    if soundId then
        local s = Instance.new("Sound")
        s.Name = "EmoteSound"
        s.SoundId = soundId
        s.Looped = loop and true or false
        for k, v in pairs(soundProps or {}) do s[k] = v end
        s.Parent = SoundService
        s:Play()
        return s
    end
end
local function stopEmote()
    local char = myChar()
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local animator = hum and hum:FindFirstChildOfClass("Animator")
    if animator then
        for _, t in ipairs(animator:GetPlayingAnimationTracks()) do t:Stop() end
    end
    local s = SoundService:FindFirstChild("EmoteSound")
    if s then s:Stop() s:Destroy() end
end
